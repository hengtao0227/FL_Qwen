/**
     * <p>Gets the greatest common divisor of the absolute value of
     * two numbers, using the "binary gcd" method which avoids
     * division and modulo operations.  See Knuth 4.5.2 algorithm B.
     * This algorithm is due to Josef Stein (1961).</p>
     *
     * @param u  a non-zero number
     * @param v  a non-zero number
     * @return the greatest common divisor, never zero
     */
Method code:
569 |     private static int greatestCommonDivisor(int u, int v) {
570 |         //if either op. is abs 0 or 1, return 1:
571 |         if (Math.abs(u) <= 1 || Math.abs(v) <= 1) {
572 |             return 1;
573 |         }
574 |         // keep u and v negative, as negative integers range down to
575 |         // -2^31, while positive numbers can only be as large as 2^31-1
576 |         // (i.e. we can't necessarily negate a negative number without
577 |         // overflow)
578 |         if (u>0) { u=-u; } // make u negative
579 |         if (v>0) { v=-v; } // make v negative
580 |         // B1. [Find power of 2]
581 |         int k=0;
582 |         while ((u&1)==0 && (v&1)==0 && k<31) { // while u and v are both even...
583 |             u/=2; v/=2; k++; // cast out twos.
584 |         }
585 |         if (k==31) {
586 |             throw new ArithmeticException("overflow: gcd is 2^31");
587 |         }
588 |         // B2. Initialize: u and v have been divided by 2^k and at least
589 |         //     one is odd.
590 |         int t = ((u&1)==1) ? v : -(u/2)/*B3*/;
591 |         // t negative: u was odd, v may be even (t replaces v)
592 |         // t positive: u was even, v is odd (t replaces u)
593 |         do {
594 |             /* assert u<0 && v<0; */
595 |             // B4/B3: cast out twos from t.
596 |             while ((t&1)==0) { // while t is even..
597 |                 t/=2; // cast out twos
598 |             }
599 |             // B5 [reset max(u,v)]
600 |             if (t>0) {
601 |                 u = -t;
602 |             } else {
603 |                 v = t;
604 |             }
605 |             // B6/B3. at this point both u and v should be odd.
606 |             t = (v - u)/2;
607 |             // |u| larger: t positive (replace u)
608 |             // |v| larger: t negative (replace v)
609 |         } while (t!=0);
610 |         return -u*(1<<k); // gcd is u*2^k
611 |     }
