     * </p>

     * @param b1 second factor of the first term
     * @param a2 first factor of the second term
     * @param b2 second factor of the second term
     * @return a<sub>1</sub>&times;b<sub>1</sub> +
     * a<sub>2</sub>&times;b<sub>2</sub>

     * @see #linearCombination(double, double, double, double, double, double, double, double)

851 |     public static double linearCombination(final double a1, final double b1,
852 |                                            final double a2, final double b2) {
853 | 
854 |         // the code below is split in many additions/subtractions that may
855 |         // appear redundant. However, they should NOT be simplified, as they
856 |         // use IEEE754 floating point arithmetic rounding properties.
857 |         // as an example, the expression "ca1 - (ca1 - a1)" is NOT the same as "a1"
858 |         // The variable naming conventions are that xyzHigh contains the most significant
859 |         // bits of xyz and xyzLow contains its least significant bits. So theoretically
860 |         // xyz is the sum xyzHigh + xyzLow, but in many cases below, this sum cannot
861 |         // be represented in only one double precision number so we preserve two numbers
862 |         // to hold it as long as we can, combining the high and low order bits together
863 |         // only at the end, after cancellation may have occurred on high order bits
864 | 
865 |         // split a1 and b1 as two 26 bits numbers
866 |         final double ca1        = SPLIT_FACTOR * a1;
867 |         final double a1High     = ca1 - (ca1 - a1);
868 |         final double a1Low      = a1 - a1High;
869 |         final double cb1        = SPLIT_FACTOR * b1;
870 |         final double b1High     = cb1 - (cb1 - b1);
871 |         final double b1Low      = b1 - b1High;
872 | 
873 |         // accurate multiplication a1 * b1
874 |         final double prod1High  = a1 * b1;
875 |         final double prod1Low   = a1Low * b1Low - (((prod1High - a1High * b1High) - a1Low * b1High) - a1High * b1Low);
876 | 
877 |         // split a2 and b2 as two 26 bits numbers
878 |         final double ca2        = SPLIT_FACTOR * a2;
879 |         final double a2High     = ca2 - (ca2 - a2);
880 |         final double a2Low      = a2 - a2High;
881 |         final double cb2        = SPLIT_FACTOR * b2;

883 |         final double b2Low      = b2 - b2High;
884 |

889 |         // accurate addition a1 * b1 + a2 * b2
890 |         final double s12High    = prod1High + prod2High;
891 |         final double s12Prime   = s12High - prod2High;
892 |         final double s12Low     = (prod2High - (s12High - s12Prime)) + (prod1High - s12Prime);
